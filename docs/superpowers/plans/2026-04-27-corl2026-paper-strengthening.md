def compute_stats(values: List[float]) -> Dict[str, float]:
    """Compute mean, std, 95% CI, and N for a list of values."""
    arr = np.array(values)
    n = len(arr)
    if n == 0:
        return {"mean": 0.0, "std": 0.0, "ci95": 0.0, "n": 0}
    mean = float(np.mean(arr))
    std = float(np.std(arr, ddof=1)) if n > 1 else 0.0
    ci95 = float(1.96 * std / np.sqrt(n)) if n > 1 else 0.0
    return {"mean": mean, "std": std, "ci95": ci95, "n": n}


def wilcoxon_test(x: List[float], y: List[float], alternative="two-sided") -> Dict[str, float]:
    """Wilcoxon signed-rank test between two samples."""
    if len(x) < 5 or len(y) < 5:
        return {"statistic": 0.0, "pvalue": 1.0, "significant": False}
    try:
        result = stats.wilcoxon(x, y, alternative=alternative)
        return {"statistic": float(result.statistic), "pvalue": float(result.pvalue), "significant": float(result.pvalue) < 0.05}
    except ValueError:
        return {"statistic": 0.0, "pvalue": 1.0, "significant": False}


def format_result_table(results: Dict[str, Dict], metrics: List[str]) -> str:
    """Format results into a readable table with mean +/- CI."""
    lines = []
    header = f"{'Method':<20}" + "".join(f"{m:<18}" for m in metrics)
    lines.append(header)
    lines.append("-" * len(header))
    for method, data in results.items():
        row = f"{method:<20}"
        for m in metrics:
            s = data.get(m, compute_stats([]))
            row += f"{s['mean']:.3f}±{s['ci95']:.3f}   "
        lines.append(row)
    return "\n".join(lines)

    """Save experiment results in standard format."""
    os.makedirs(output_dir, exist_ok=True)

    with open(json_path, 'w') as f:

#!/usr/bin/env python
"""Build dense topological maps for any CARLA town.

Wraps corl_2026_map_town02.py logic to work with arbitrary towns.

Usage:
    python -m experiments.map_town --town Town01 --output-dir corl-2026/results/town01_map
    python -m experiments.map_town --town Town03 --output-dir corl-2026/results/town03_map
"""
import os
import sys
import argparse

# Add repo root to path for imports
sys.path.insert(0, os.path.join(os.path.dirname(__file__), '..', '..'))

def build_town_map(town: str, output_dir: str, sampling_resolution: float = 2.0,
                   image_size: tuple = (256, 256), capture_interval: float = 0.5,
                   n_clusters: int = 64, skip_existing: bool = True):
    """Build dense topological map for a given CARLA town.

    Calls the existing corl_2026_map_town02.py main() with overridden args.
    """
    import carla

    # Import the existing builder
    from corl_2026_map_town02 import CarlaSyncMode, TownMapBuilder, MapDataCollector

    # Connect to CARLA
    client = carla.Client('localhost', 2000)
    client.set_timeout(20.0)
    world = client.load_world(town)
    m = world.get_map()

    os.makedirs(output_dir, exist_ok=True)

    # Check if already built
    if skip_existing and os.path.exists(os.path.join(output_dir, "map_metadata.json")):
        print(f"Map for {town} already exists at {output_dir}, skipping.")
        return output_dir

    print(f"Building map for {town} -> {output_dir}")

    # Collect map data
    builder = TownMapBuilder(m, sampling_resolution=sampling_resolution)
    ordered_waypoints = builder.get_ordered_waypoints()

    collector = MapDataCollector(
        save_dir=output_dir,
        image_size=image_size,
        capture_interval=capture_interval,
    )

    # Spawn vehicle and sensors
    blueprint_library = world.get_blueprint_library()
    vehicle_bp = blueprint_library.filter('vehicle.*')[0]

    start_pose = m.get_spawn_points()[0].transform if m.get_spawn_points() else carla.Transform()
    vehicle = world.spawn_actor(vehicle_bp, start_pose)
    vehicle.set_simulate_physics(False)

    camera_rgb = world.spawn_actor(
        blueprint_library.find('sensor.camera.rgb'),
        carla.Transform(carla.Location(x=1.0, z=2.8), carla.Rotation(pitch=-15)),
        attach_to=vehicle
    )

    try:
        with CarlaSyncMode(world, camera_rgb, fps=30) as sync_mode:
            for i, waypoint in enumerate(ordered_waypoints):
                vehicle.set_transform(waypoint.transform)
                snapshot, image_rgb, _ = sync_mode.tick(timeout=2.0)
                collector.save_image(image_rgb, waypoint)

                if (i + 1) % 100 == 0:
                    print(f"  {town}: {i+1}/{len(ordered_waypoints)} waypoints processed")

        collector.save_map_data()
        print(f"Map for {town} built: {len(collector.nodes_data)} nodes")

    finally:
        vehicle.destroy()
        camera_rgb.destroy()

    return output_dir


def main():
    parser = argparse.ArgumentParser(description="Build topological map for any CARLA town")
    parser.add_argument('--town', type=str, required=True)
    parser.add_argument('--output-dir', type=str, required=True)
    parser.add_argument('--sampling-resolution', type=float, default=2.0)
    parser.add_argument('--image-size', type=int, nargs=2, default=[256, 256])
    parser.add_argument('--capture-interval', type=float, default=0.5)
    parser.add_argument('--n-clusters', type=int, default=64)
    parser.add_argument('--skip-existing', action='store_true')
    parser.add_argument('--carla-port', type=int, default=2000)
    args = parser.parse_args()

    build_town_map(
        town=args.town,
        output_dir=args.output_dir,
        sampling_resolution=args.sampling_resolution,
        image_size=tuple(args.image_size),
        capture_interval=args.capture_interval,
        n_clusters=args.n_clusters,
        skip_existing=args.skip_existing,
    )


if __name__ == "__main__":
    main()

#!/usr/bin/env python
"""Create benchmark trajectories for any CARLA town.

Extends corl_2026_create_trajectories.py to support arbitrary towns
and configurable trajectory counts.

Usage:
    python -m experiments.trajectory_creator --town Town01 --num-trajectories 50
"""
import os
import sys
import argparse

sys.path.insert(0, os.path.join(os.path.dirname(__file__), '..', '..'))


def create_trajectories(town: str, map_dir: str, output_dir: str,
                       num_trajectories: int = 50, seed: int = 42):
    """Create benchmark trajectories for a given town."""
    import carla
    from corl_2026_create_trajectories import generate_trajectory_pairs, create_trajectory_from_map

    client = carla.Client('localhost', 2000)
    client.set_timeout(20.0)
    world = client.load_world(town)
    wmap = world.get_map()

    os.makedirs(output_dir, exist_ok=True)

    # Check existing
    info_path = os.path.join(output_dir, "benchmark_info.json")
    if os.path.exists(info_path):
        with open(info_path, 'r') as f:
            existing = json.load(f) if os.path.getsize(info_path) > 0 else {}
        if existing.get("num_trajectories", 0) >= num_trajectories:
            print(f"Trajectories for {town} already exist ({existing['num_trajectories']}), skipping.")
            return output_dir

    # Load map data
    with open(os.path.join(map_dir, "map_data.pkl"), "rb") as f:
        map_data = pickle.load(f)

    print(f"Creating {num_trajectories} trajectories for {town}")

    # Generate trajectory pairs
    traj_pairs = generate_trajectory_pairs(wmap, num_trajectories, seed=seed)

    # Create each trajectory
    trajectories_info = []
    for i, traj_info in enumerate(traj_pairs):
        traj_id = f"trajectory_{i:02d}"
        traj_dir = os.path.join(output_dir, traj_id)
        os.makedirs(traj_dir, exist_ok=True)

        create_trajectory_from_map(traj_info, map_data, map_dir, traj_dir)
        trajectories_info.append({
            "trajectory_id": i,
            "path": traj_dir,
            "start": traj_info.get("start", []),
            "goal": traj_info.get("goal", []),
            "route_length": traj_info.get("route_length", 0),
        })

        if (i + 1) % 10 == 0:
            print(f"  Created {i+1}/{num_trajectories} trajectories")

    # Save benchmark info
    benchmark_info = {
        "town": town,
        "num_trajectories": len(trajectories_info),
        "trajectories": trajectories_info,
        "seed": seed,
    }
    with open(info_path, 'w') as f:
        json.dump(benchmark_info, f, indent=2)

    print(f"Created {len(trajectories_info)} trajectories for {town}")
    return output_dir


def main():
    import json
    import pickle

    parser = argparse.ArgumentParser(description="Create benchmark trajectories for any CARLA town")
    parser.add_argument('--town', type=str, required=True)
    parser.add_argument('--map-dir', type=str, required=True)
    parser.add_argument('--output-dir', type=str, required=True)
    parser.add_argument('--num-trajectories', type=int, default=50)
    parser.add_argument('--seed', type=int, default=42)
    parser.add_argument('--carla-port', type=int, default=2000)
    args = parser.parse_args()

    create_trajectories(
        town=args.town,
        map_dir=args.map_dir,
        output_dir=args.output_dir,
        num_trajectories=args.num_trajectories,
        seed=args.seed,
    )


if __name__ == "__main__":
    main()

#!/usr/bin/env python
"""Experiment 1: Scaled Evaluation on 50+ trajectories.

Extends corl_2026_evaluate_image_goal.py with:
  - 50+ trajectories in Town02
  - Multiple seeds for stochastic methods
  - Statistical tests (Wilcoxon signed-rank)
  - Error bars and confidence intervals

Usage:
    python -m experiments.experiment1_scaled_eval --model all --town Town02 \
        --num-trajectories 50 --num-seeds 3

Requires: CARLA server running on port 2000
"""

import sys
import json
import pickle
import argparse
from collections import defaultdict

import numpy as np
import tqdm

sys.path.insert(0, os.path.join(os.path.dirname(__file__), '..', '..'))

os.environ['XLA_FLAGS'] = '--xla_gpu_enable_command_buffer='
os.environ['XLA_PYTHON_CLIENT_PREALLOCATE'] = 'false'
os.environ['XLA_PYTHON_CLIENT_MEM_FRACTION'] = '.20'
os.environ['XLA_PYTHON_CLIENT_ALLOCATOR'] = 'platform'

from experiments.common import (
    compute_stats, wilcoxon_test, save_results, load_results,
    make_base_parser, format_result_table,
    TERMINATION_BUDGET, MAX_EPISODE_STEPS, SEED, SUBGOAL_REACH_DIST,
    MODEL_CHECKPOINTS, DRQ_IG_CHECKPOINT, DEFAULT_BENCHMARK_DIR, DEFAULT_MAP_DIR,
)
from experiments.trajectory_creator import create_trajectories


def run_scaled_eval(args):
    """Run the scaled evaluation experiment."""
    # Import CARLA-dependent modules only when running
    from corl_2026_evaluate_image_goal import (
        DenseMapNavigator, load_drq_ig_agent,
        evaluate_drq_ig_trajectory, evaluate_baseline_trajectory,
    )
    from rlib_integration.carla_env import CarlaEnv
    from jaxrl2.wrappers.record_statistics import RecordEpisodeStatistics
    from jaxrl2.wrappers.timelimit import TimeLimit
    from src.configs.eval_env_config_image_goal import config as ig_config

    output_dir = os.path.join(args.output_dir, "scaled_eval")
    os.makedirs(output_dir, exist_ok=True)

    # Step 1: Ensure we have enough trajectories
    benchmark_dir = args.benchmark_dir
    info_path = os.path.join(benchmark_dir, "benchmark_info.json")
    if os.path.exists(info_path):
        with open(info_path, 'r') as f:
            benchmark_info = json.load(f)
    else:
        benchmark_info = {"trajectories": [], "town": args.town}

    existing_count = len(benchmark_info.get("trajectories", []))
    if existing_count < args.num_trajectories:
        print(f"Need {args.num_trajectories} trajectories, have {existing_count}. Creating more...")
        create_trajectories(
            town=args.town,
            map_dir=args.map_dir,
            output_dir=benchmark_dir,
            num_trajectories=args.num_trajectories,
            seed=args.seed,
        )
        with open(info_path, 'r') as f:
            benchmark_info = json.load(f)

    # Step 2: Load navigator
    nav = DenseMapNavigator(args.map_dir, localization_mode=args.localization_mode)

    # Step 3: Setup environment
    ig_config["env_config"]["carla"]["start_server"] = False
    ig_config["env_config"]["carla"]["port"] = args.carla_port
    ig_config["env_config"]["carla"]["town"] = args.town
    env = CarlaEnv(config=ig_config)
    env.experiment.eval_mode = True
    env.experiment.map_mode = False
    env = TimeLimit(env, max_episode_steps=MAX_EPISODE_STEPS)
    env = RecordEpisodeStatistics(env)

    # Step 4: Evaluate each model
    models_to_eval = ['drq_ig', 'nomad', 'gnm', 'vint'] if args.model == 'all' else [args.model]
    all_results = {}

    for model_type in models_to_eval:
        print(f"\n{'='*60}")
        print(f"Evaluating {model_type.upper()} on {len(benchmark_info['trajectories'])} trajectories")
        print(f"{'='*60}")

        if model_type == 'drq_ig':
            agent = load_drq_ig_agent(env, DRQ_IG_CHECKPOINT, SEED)

        per_traj_results = []
        for traj_info in tqdm.tqdm(benchmark_info["trajectories"][:args.num_trajectories],
                                    desc=f"{model_type}"):
            traj_id = traj_info["trajectory_id"]
            traj_path = traj_info["path"]

            try:
                if model_type == 'drq_ig':
                    result = evaluate_drq_ig_trajectory(
                        env, agent, nav, traj_path, traj_id, MAX_EPISODE_STEPS
                    )
                else:
                    result = evaluate_baseline_trajectory(
                        env, model_type, nav, traj_path, traj_id, MAX_EPISODE_STEPS
                    )
                per_traj_results.append(result)
            except Exception as e:
                print(f"  ERROR on traj {traj_id}: {e}")
                per_traj_results.append({
                    "trajectory_id": traj_id, "model": model_type,
                    "error": str(e), "success": 0.0, "mean_spl": 0.0,
                    "mean_distance": 0.0, "path_length": 0,

def _detect_junctions_from_heading(nodes_data):
    """Detect junctions by significant heading changes between consecutive nodes."""
    junction_indices = []
    heading_threshold = 30.0  # degrees
    for i in range(1, len(nodes_data) - 1):
        prev_yaw = nodes_data[i-1].get('yaw', 0)
        curr_yaw = nodes_data[i].get('yaw', 0)
        next_yaw = nodes_data[i+1].get('yaw', 0)
        delta1 = abs(curr_yaw - prev_yaw) % 360
        delta2 = abs(next_yaw - curr_yaw) % 360
        if delta1 > 180: delta1 = 360 - delta1
        if delta2 > 180: delta2 = 360 - delta2
        if delta1 > heading_threshold or delta2 > heading_threshold:
            junction_indices.append(i)
    return junction_indices


def build_sparse_graph_from_subset(full_graph, selected_indices, nodes_data):
    """Build a sparse graph from a subset of nodes, connecting consecutive selected nodes."""
    G = nx.Graph()
    for idx in selected_indices:
        G.add_node(idx, **nodes_data[idx])

    # Connect consecutive selected nodes along the original route
    for i in range(len(selected_indices) - 1):
        a, b = selected_indices[i], selected_indices[i + 1]
        pos_a = np.array([nodes_data[a]['x'], nodes_data[a]['y']])
        pos_b = np.array([nodes_data[b]['x'], nodes_data[b]['y']])
        dist = np.linalg.norm(pos_a - pos_b)
        G.add_edge(a, b, weight=dist)

    # Add loop closures from full graph where both endpoints are in selected set
    selected_set = set(selected_indices)
    for u, v in full_graph.edges():
        if u in selected_set and v in selected_set and not G.has_edge(u, v):
            dist = np.linalg.norm(
                np.array([nodes_data[u]['x'], nodes_data[u]['y']]) -
                np.array([nodes_data[v]['x'], nodes_data[v]['y']])
            )
            G.add_edge(u, v, weight=dist)

    return G

                              for i in self.selected_indices])

import sys

# MAD spike detection implementation
class MADSpikeDetector:
    """Median Absolute Deviation based spike detector.

    Parameters:
        window (int): Rolling buffer size for median computation
        threshold (float): Number of MADs above median to trigger detection
        lockout (int): Minimum steps between detections
    """
    def __init__(self, window=200, threshold=3.0, lockout=20):
        self.window = window
        self.threshold = threshold
        self.lockout = lockout
        self.buffer = []
        self.last_detection = -lockout  # Allow immediate detection
        self.step = 0

    def update(self, value):
        """Process a new value, return True if spike detected."""
        self.buffer.append(value)
        if len(self.buffer) > self.window:
            self.buffer = self.buffer[-self.window:]

        if len(self.buffer) < 10:
            self.step += 1
            return False

        median = np.median(self.buffer)
        mad = np.median(np.abs(np.array(self.buffer) - median))

        if mad > 0 and (value - median) > self.threshold * mad:
            if self.step - self.last_detection >= self.lockout:
                self.last_detection = self.step
                self.step += 1
                return True

        self.step += 1
        return False

def _collect_entropy_profiles(env, agent, n_episodes=5):
    """Collect policy entropy during exploration episodes."""
    import jax
    profiles = []
    for ep in range(n_episodes):
        obs, info = env.reset()
        done = False
        profile = []
        while not done:
            # Get action distribution from agent
            dist = agent._actor.apply_fn(agent._actor.params, obs)
            entropy = -np.sum(dist.log_prob(dist.sample(seed=jax.random.PRNGKey(0))) * np.exp(dist.log_prob(dist.sample(seed=jax.random.PRNGKey(1)))))
            # Simplified: use stddev as proxy for entropy
            profile.append(float(np.mean(agent.eval_action_std(obs))))
            action = agent.eval_actions(obs)
            obs, reward, done, truncated, info = env.step(action)
            done = done or truncated
        profiles.append(profile)
    return profiles


def _detect_nodes_from_entropy(entropy_profiles, params):
    """Apply MAD spike detection to entropy profiles and return detected step indices."""
    all_detected = set()
    for profile in entropy_profiles:
        detector = MADSpikeDetector(
            window=params["window"],
            threshold=params["threshold"],
            lockout=params["lockout"],
        )
        for step, value in enumerate(profile):
            if detector.update(value):
                all_detected.add(step)
    return sorted(all_detected)


def _compute_junction_accuracy(detected_steps, nodes_data):
    """Compute fraction of detected nodes that are at junctions."""
    if not detected_steps:
        return 0.0
    junction_count = 0
    for step in detected_steps:
        if step < len(nodes_data):
            node = nodes_data[step]
            if node.get('is_junction', False):
                junction_count += 1
    return junction_count / len(detected_steps)

                        help="Number of exploration episodes for entropy collection")

    run_spike_detection_ablation(args)

\appendix
\section{Formal Proofs}

\subsection{Proof of Proposition 1: Reachability Preservation}

\begin{proposition}[Reachability Preservation]
\label{prop:reachability}
Under assumptions that (A1) the exploration policy $\pi_e$ is ergodic over the reachable state space $\calS_R \subseteq \calS$, and (A2) the graph $\calM = \calG(\calV, \calE)$ connects sequentially detected decision states along each exploration trajectory, the sparse topological map $\calM$ preserves reachability between any two decision states that are connected by a path in $\calS_R$.
\end{proposition}

\begin{proof}
Let $d_i, d_j \in \calV$ be two decision states detected during exploration, and let there exist a path $p = (s_0, s_1, \ldots, s_K)$ in $\calS_R$ from $d_i$ to $d_j$.

By (A1), the exploration policy $\pi_e$ visits each state in $\calS_R$ with positive probability. In particular, for each junction state $s_k \in p$ that is a decision state, there exists an exploration trajectory $\tau$ that passes through $s_k$.

By (A2), when $\pi_e$ traverses the path from $d_i$ to $d_j$, the algorithm detects decision states along this trajectory and adds them to $\calV$. The edges $\calE$ connect consecutively detected decision states. Since the path $p$ contains only states in $\calS_R$, and $\pi_e$ is ergodic over $\calS_R$, the exploration will eventually traverse a path that includes all decision states along $p$.

More formally, let $\{d_{i_1}, d_{i_2}, \ldots, d_{i_m}\} \subseteq \calV$ be the decision states detected along a single exploration trajectory that traverses from $d_i$ to $d_j$. By construction, consecutive decision states in this sequence are connected by edges in $\calE$. Therefore, there exists a path $d_{i_1} \to d_{i_2} \to \cdots \to d_{i_m}$ in $\calM$, which implies $d_i$ and $d_j$ are connected in $\calM$.

This holds for any pair of decision states connected in $\calS_R$, completing the proof.
\end{proof}

\subsection{Proof of Proposition 2: Grid-World Correctness}

\begin{proposition}[Grid-World Junction Detection]
\label{prop:grid_world}
Under the lane-following policy $\pi_l$ defined in Eq.~(1)--(2) with $\lambda_\text{lane} = \lambda_\text{coll} = 5$:
\begin{enumerate}[(a)]
    \item At non-junction states where only forward motion is feasible, $H(\pi_l(\cdot|s)) \leq \log 3 \cdot e^{-5} \approx 0.01$.
    \item At T-junctions where two actions are feasible, $H(\pi_l(\cdot|s)) \geq \log 2 \cdot (1 - 2e^{-5}) \approx 0.69$.
    \item The entropy-based detection algorithm with sensitivity $k \geq 2$ correctly identifies all junction states.
\end{enumerate}
\end{proposition}

\begin{proof}
\textbf{Part (a):} At a non-junction state $s$, the only feasible action is $\bfa = \text{forward}$. For any other action $\bfa' \in \{\text{rotate left}, \text{rotate right}\}$, taking $\bfa'$ would result in either a lane invasion ($\indicator_\text{lane} = 1$) or a collision ($\indicator_\text{coll} = 1$). Thus:
\begin{align}
H_l(s, \text{forward}) &= t \\
H_l(s, \bfa') &= t - \lambda_\text{lane} \cdot \indicator_\text{lane} - \lambda_\text{coll} \cdot \indicator_\text{coll} = t - 5
\end{align}

The softmax policy gives:
\begin{align}
\pi_l(\text{forward}|s) &= \frac{e^t}{e^t + e^{t-5} + e^{t-5}} = \frac{1}{1 + 2e^{-5}} \approx 0.9933 \\
\pi_l(\bfa'|s) &= \frac{e^{-5}}{1 + 2e^{-5}} \approx 0.0033
\end{align}

The entropy is:
\begin{align}
H(\pi_l(\cdot|s)) &= -\sum_{\bfa} \pi_l(\bfa|s) \log \pi_l(\bfa|s) \\
&\leq \log 3 \cdot \max_\bfa \pi_l(\bfa|s) \cdot (1 - \pi_l(\bfa|s)) \\
&\leq \log 3 \cdot e^{-5} \approx 0.006
\end{align}

\textbf{Part (b):} At a T-junction state $s$ with two feasible actions (e.g., forward and rotate left), both feasible actions have $H_l = t$ while the infeasible action has $H_l = t - 5$:
\begin{align}
\pi_l(\text{forward}|s) = \pi_l(\text{rotate left}|s) &= \frac{e^t}{2e^t + e^{t-5}} = \frac{1}{2 + e^{-5}} \approx 0.4997 \\
\pi_l(\text{rotate right}|s) &= \frac{e^{-5}}{2 + e^{-5}} \approx 0.0007
\end{align}

The entropy is:
\begin{align}
H(\pi_l(\cdot|s)) &\approx 2 \cdot 0.4997 \cdot \log(1/0.4997) + 0.0007 \cdot \log(1/0.0007) \\
&\approx 0.693 \cdot (1 - 2e^{-5}) \approx 0.693
\end{align}

\textbf{Part (c):} The entropy ratio between junction and non-junction states is:
\begin{align}
\frac{H_\text{junction}}{H_\text{non-junction}} \approx \frac{0.693}{0.006} \approx 115.5
\end{align}

With a MAD-based threshold of $k \geq 2$, any detection algorithm that identifies outliers at $2\sigma$ above the median will clearly separate the junction states (entropy $\sim 0.69$) from non-junction states (entropy $\sim 0.006$), as the gap exceeds two orders of magnitude. Therefore, all junction states are correctly identified.
\end{proof}

\subsection{Proof of Theorem 1: Information-Theoretic Bound}

\begin{theorem}[Minimum Decision States]
\label{thm:info_bound}
For a navigation task with success rate $\varepsilon$ over graph $\calM = \calG(\calV, \calE)$, the minimum number of decision states satisfies:
\begin{align}
|\calV^*| \geq |\calD| \geq \frac{I(\calS; \calA)}{\max_{s \in \calD} H(\pi(\cdot|s))}
\end{align}
where $\calD$ is the set of decision states, $I(\calS; \calA)$ is the mutual information between states and actions, and $H(\pi(\cdot|s))$ is the policy entropy at state $s$.
\end{theorem}

\begin{proof}
The mutual information between states and actions for the navigation policy can be decomposed as:
\begin{align}
I(\calS; \calA) &= H(\calA) - H(\calA|\calS) \\
&= \sum_{s \in \calS} p(s) \sum_{a \in \calA} \pi(a|s) \log \frac{\pi(a|s)}{p(a)} 
\end{align}

where $p(a) = \sum_s p(s)\pi(a|s)$ is the marginal action distribution.

For navigation to succeed with rate $\varepsilon$, the agent must be able to make distinct navigation decisions at each decision state. This requires that the mutual information between the decision states and the actions is at least $I(\calD; \calA) \geq \varepsilon \cdot I(\calS; \calA)$, since the decision states are the states where action selection matters for navigation.

By the chain rule of mutual information and the data processing inequality:
\begin{align}
I(\calD; \calA) &\leq \sum_{d \in \calD} H(\pi(\cdot|d)) \\
&\leq |\calD| \cdot \max_{d \in \calD} H(\pi(\cdot|d))
\end{align}

Therefore:
\begin{align}
|\calD| &\geq \frac{I(\calD; \calA)}{\max_{d \in \calD} H(\pi(\cdot|d))} \\
&\geq \frac{\varepsilon \cdot I(\calS; \calA)}{\max_{d \in \calD} H(\pi(\cdot|d))}
\end{align}

Setting $\varepsilon = 1$ for the bound on the minimum number of decision states needed for perfect navigation gives the result.
\end{proof}

\subsection{SAC Entropy as Decision-State Proxy}

\label{app:sac_entropy}

In SAC, the policy is parameterized as a squashed Gaussian: $\bfa = \tanh(\bfmu + \bfSigma^{1/2} \bfepsilon)$ where $\bfepsilon \sim \calN(\bf0, \bfI)$.

The \textit{squashed} entropy $H_\text{squash}(\pi_\phi) = -\mathbb{E}_{\pi_\phi}[\log \pi_\phi(\bfa|\bfs)]$ is intractable because the change-of-variables through $\tanh$ produces a complex density. However, the \textit{unsquashed} entropy has a closed form:
\begin{align}
H(\pi_\phi) = \frac{k}{2}\log(2\pi e) + \frac{1}{2}\log \det \bfSigma_\pi
\end{align}

where $k$ is the action dimension and $\bfSigma_\pi$ is the covariance matrix.

We use the unsquashed entropy as a proxy for decision-state detection for three reasons:
\begin{enumerate}
    \item \textbf{Monotonicity}: The unsquashed entropy is monotonically related to the ``decision complexity'' at a state. When $\bfSigma_\pi$ is large (high uncertainty), the unsquashed entropy is high, indicating the policy has multiple viable actions---exactly the signature of a decision state.
    \item \textbf{Computational simplicity}: The unsquashed entropy is a closed-form function of the covariance matrix, which is already computed by the actor network during inference. No additional computation is required.
    \item \textbf{Connection to Maximum Entropy RL}: SAC's objective includes $\alpha \cdot H(\pi_\phi)$ as an entropy bonus. The unsquashed entropy directly measures the exploration-exploitation trade-off that the policy is optimizing. High unsquashed entropy at a state means the policy is ``exploring''---which, in the context of a trained low-level policy, indicates that no single action is clearly preferred, i.e., a decision state.
\end{enumerate}

% Future Directions for Sparse Topological Mapping
% To be included as supplementary material or discussion section

\section*{Future Directions}

\paragraph{Online Map Refinement}
Our current approach builds the topological map during a single exploration phase. An exciting direction is online map refinement: as the agent navigates using the map, it can detect when localization fails or when the navigation plan leads to a dead end, and add new nodes to the map on-the-fly. This combines exploration and exploitation within a single lifelong-learning framework.

\paragraph{Multi-Modal Decision States}
Our current definition of decision states relies solely on the entropy of the action distribution. However, decision states could be enriched with additional modalities:
\begin{itemize}
    \item \textbf{Visual novelty}: States where the visual observation is significantly different from previously seen observations (novelty detection).
    \item \textbf{Collision history}: States where the agent has previously collided or left the lane, indicating a location that requires careful navigation.
    \item \textbf{Multi-agent uncertainty}: In settings with multiple agents, states where other agents' behavior introduces additional uncertainty.
\end{itemize}
A multi-modal decision-state detector could be more robust than entropy alone.

\paragraph{Hierarchical Map Structures}
Our current map is flat---all nodes are at the same level. For very large environments, a hierarchical map where high-level nodes represent regions (neighborhoods, building floors) and low-level nodes represent specific decision points within regions could enable even sparser representations. This naturally extends the two-level hierarchy (low-level policy + topological map) to a multi-level hierarchy.

\paragraph{Active Exploration for Mapping}
Our exploration policy is currently passive---it follows lanes and takes random turns at junctions. An active exploration strategy could prioritize visiting regions of the environment that are likely to contain decision states, reducing the exploration time needed to build a complete topological map. This could be formulated as a Bayesian optimization problem where the objective is to maximize map coverage with minimum exploration steps.

\paragraph{Transfer Across Embodiments}
Our low-level policy is trained for a specific vehicle in CARLA. An interesting direction is to study how the topological map changes when the low-level policy is replaced with a different embodiment (e.g., a walking robot vs. a car). Decision states for a car (junctions) differ from decision states for a walking robot (elevators, staircases). This suggests that the topological map is embodiment-dependent, and a unified mapping framework should account for this.

\paragraph{Integration with Foundation Models}
Recent vision-language foundation models (e.g., CLIP, GPT-4V) could provide richer goal specifications (natural language, object descriptions) and more robust localization. Integrating these models with our sparse topological map could enable more flexible and generalizable navigation systems.

\paragraph{Theoretical Extensions}
Our information-theoretic bound (Theorem 1) provides a lower bound on the number of decision states. A complementary upper bound---showing that our algorithm achieves a map size within a constant factor of optimal---would strengthen the theoretical contribution. This could be approached through competitive analysis of online algorithms, where the ``optimal'' map is one built with full knowledge of the environment.

\paragraph{Real-World Deployment}
The most important next step is real-world deployment. Our CARLA experiments demonstrate the core idea, but real-world validation is essential. Key challenges include: (1) sim-to-real transfer of the low-level policy, (2) robustness to dynamic obstacles and weather changes, (3) computational constraints on embedded platforms, and (4) safety guarantees for the navigation system.
